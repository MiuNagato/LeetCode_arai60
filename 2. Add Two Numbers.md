```python
class Solution:

        stack = []
        for(i, max(l1,l2).length, i++):
            value = l1(i)+l2(i)
            if value >= 10:
                l1(i+1)+=1
                stack.append(value)
                value = None
```

```python
class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode], carry :int=0) -> Optional[ListNode]:
        # 終了条件 処理なしの場合は先にreturnすべきか
        if l1 is None and l2 is None and carry == 0:
            return None
        # 番兵
        if l1 is None:
            l1 = ListNode(0) # l1を直接書き替え
        if l2 is None:
            l2 = ListNode(0)
        sum = l1.val + l2.val + carry
        # 一桁目がNode.val 2桁目がcarry
        carry,value = divmod(sum,10) # 商 → carry 余り → value 一括で取得
        head = ListNode(value)
        head.next = self.addTwoNumbers(l1.next,l2.next,carry)
        return head
```

```python
class Solution:
    def addTwoNumbers(
        self, l1: Optional[ListNode], l2: Optional[ListNode]
    ) -> Optional[ListNode]:
        dummy = ListNode()
        current = dummy
        carry = 0
        while l1 or l2 or carry:
            l1_val = 0
            if l1:
                l1_val = l1.val # ほかとは異なり略さずに書いたパターン。略した方が分かりやすいこともあるんだな。
                l1 = l1.next # 次の準備をここでする。なるべくwhileの末でやりたくないか？
            l2_val = 0
            if l2:
                l2_val = l2.val
                l2 = l2.next
            current_sum = l1_val + l2_val + carry
            carry = current_sum // 10
            current.next = ListNode(val=current_sum % 10)
            current = current.next
        return dummy.next
```
